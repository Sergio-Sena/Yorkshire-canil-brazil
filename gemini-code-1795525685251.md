## 3. Ajustes Recomendados no Código (`js/main.js`)

Substituir o bloco de rastreamento de links do WhatsApp e chamar `bindWhatsAppLinks()` após a renderização dinâmica das seções de vídeo em `loadMedia()`:

```javascript
// Função centralizada de rastreamento e UTM para TODOS os links de WhatsApp (estáticos e dinâmicos)
function bindWhatsAppLinks() {
  const params = new URLSearchParams(window.location.search);
  const utmSource = params.get("utm_source");
  const utmMedium = params.get("utm_medium");
  const utmCampaign = params.get("utm_campaign") || "";
  const gclid = params.get("gclid");
  const isPago = Boolean(gclid || (utmSource === "google" && utmMedium === "cpc"));

  if (isPago) {
    sessionStorage.setItem("ycb_origem", "pago");
    if (gclid) sessionStorage.setItem("ycb_gclid", gclid);
    if (utmCampaign) sessionStorage.setItem("ycb_campanha", utmCampaign);
  }

  const origemFinal = sessionStorage.getItem("ycb_origem") === "pago" ? "pago" : "organico";

  document.querySelectorAll('a[href*="wa.me/5511977118201"]').forEach((link) => {
    if (link.dataset.wppBound === "true" || link.closest("#form-lead")) return;
    link.dataset.wppBound = "true";

    const secao =
      link.closest("section")?.id ||
      (link.classList.contains("whatsapp-float") ? "flutuante" : "rodape");
    const btnTexto =
      link.innerText?.trim().slice(0, 40) ||
      link.getAttribute("aria-label") ||
      "link";

    const urlObj = new URL(link.href);
    const textoAtual = urlObj.searchParams.get("text");

    if (!textoAtual) {
      const prefixo =
        origemFinal === "pago"
          ? "Olá! Vi vocês no Google"
          : "Olá! Encontrei vocês pelo site";
      const complemento =
        secao === "faq"
          ? " e gostaria de saber valores e disponibilidade dos filhotes de Yorkshire."
          : " e tenho interesse em um filhote de Yorkshire.";
      link.href = `[https://wa.me/5511977118201?text=$](https://wa.me/5511977118201?text=$){encodeURIComponent(prefixo + complemento)}`;
    }

    link.addEventListener("click", function (e) {
      const href = this.href;
      const novaAba = this.target === "_blank";

      e.preventDefault();
      let disparado = false;

      const navegar = () => {
        if (disparado) return;
        disparado = true;
        if (novaAba) {
          window.open(href, "_blank", "noopener");
        } else {
          window.location.href = href;
        }
      };

      window.dataLayer = window.dataLayer || [];
      dataLayer.push({
        event: "whatsapp_click",
        wpp_origem: origemFinal,
        wpp_secao: secao,
        wpp_btn: btnTexto,
        eventCallback: navegar,
        eventTimeout: 600
      });

      setTimeout(navegar, 600);
    });
  });
}

bindWhatsAppLinks();
